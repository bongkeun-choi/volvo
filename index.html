function renderSaeOemStyle(d) { const renderBullets = (arr) => { if (!arr || arr.length === 0 || (arr.length === 1 && arr[0] === '없음')) return `
○없음
`; return arr.map(t => `
○${t}
`).join(''); }; let fmiTablesHtml = ''; (d.fmiList || []).forEach(f => { const actionText = f.actions || d.displayMsg || d.dtc; fmiTablesHtml += `
고장 모드
FMI ${f.fmi}
 탐지 조건: ${f.cond || '-'}
증상:
${renderBullets(splitToList(f.symptom))}
가능한 이유:
${renderBullets(splitToList(f.causes))}
조치     ${actionText}
`; }); const standardHtml = d.standard ? `${d.standard}` : ''; return `
${d.dtc} ${d.displayMsg ? '/ ' + d.displayMsg : ''}
일련 번호에 대해 유효
모델 종류: 생산 사업장 일련 번호 시작 일련 번호 종료
${d.modelType || 'EW145B Volvo'} ${d.factory || '-'} ${d.serialStart || '-'} ${d.serialEnd || '-'}
SAE 코드
MID/컨트롤 유니트${d.ecu || '-'}
PID/PPID/SID/PSID${d.paramName || '-'}
장비 디스플레이 메시지
텍스트 프레젠테이션${d.displayMsg || d.dtc}
기준${standardHtml}
고장 모드 진단
${fmiTablesHtml}
Guided Diagnostics
`; }
