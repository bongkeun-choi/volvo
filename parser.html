function processPdf(file) { return new Promise((resolve, reject) => { const reader = new FileReader(); reader.onload = async function() { try { const typedarray = new Uint8Array(this.result); const pdf = await pdfjsLib.getDocument({ data: typedarray, disableFontFace: true }).promise; let fullText = ''; for (let p = 1; p <= pdf.numPages; p++) { const page = await pdf.getPage(p), content = await page.getTextContent(); const items = content.items.map(it => ({ str: it.str.trim(), y: it.transform[5], x: it.transform[4] })).filter(it => it.str.length > 0); items.sort((a, b) => (b.y - a.y) || (a.x - b.x)); let lines = [], currY = null, currLine = []; for (let it of items) { if (currY === null || Math.abs(it.y - currY) > 4) { if (currLine.length) lines.push(currLine.join(' ')); currLine = [it.str]; currY = it.y; } else currLine.push(it.str); } if (currLine.length) lines.push(currLine.join(' ')); fullText += lines.join('\n') + '\n'; } const clean = (s) => s ? s.replace(/\r?\n+/g, ' ').replace(/\s+/g, ' ').trim() : ''; const cleanBullets = (s) => { if (!s) return '없음'; const lines = s.split(/[\r\n¡○•●\t]+/).map(l => l.replace(/^[:\s\-O0]+/, '').trim()).filter(l => l && l !== '없음'); return lines.length > 0 ? lines.join(' | ') : '없음'; }; const isSae = /MID\s*\d+|PID\s*\d+|SID\s*\d+|FMI\s*\d+/i.test(fullText); if (isSae) { const midMatch = fullText.match(/MID\s*(\d+)/i); const pidMatch = fullText.match(/(PID|PPID|SID|PSID)\s*(\d+)/i); const mid = midMatch ? midMatch[1] : ''; const pidType = pidMatch ? pidMatch[1] : 'PID'; const pid = pidMatch ? pidMatch[2] : ''; const dtc = (mid && pid) ? `MID${mid} ${pidType}${pid}` : 'MID128 PID100'; let disp = (mid && pid) ? `${mid} ${pidType}${pid}` : dtc; const mDisp = fullText.match(/텍스트\s*프레젠테이션\s*([^\r\n]+)/); if (mDisp) disp = clean(mDisp[1]); // 일련 번호 및 모델 추출 let modelType = 'EW145B Volvo', factory = '', serialStart = '', serialEnd = ''; const serialBlock = fullText.match(/일련\s*번호에\s*대해\s*유효[\s\S]*?(?=SAE|장비|기준|$)/i); if (serialBlock) { const lines = serialBlock[0].split('\n').map(l => l.trim()).filter(l => l.length > 0); const dataLines = lines.filter(l => !l.includes('일련 번호에 대해 유효') && !l.includes('모델 종류') && !l.includes('생산 사업장') && !l.includes('일련 번호')); if (dataLines.length > 0) { const rowText = dataLines.join(' ').replace(/\s+/g, ' ').trim(); const mFull = rowText.match(/^(.*?)\s+([A-Za-z가-힣]+)\s+(\d{4,8})\s+(\d{4,8})$/); if (mFull) { modelType = mFull[1]; factory = mFull[2]; serialStart = mFull[3]; serialEnd = mFull[4]; } else { const parts = rowText.split(/\s+/); if (parts.length >= 4 && /^\d+$/.test(parts[parts.length - 1]) && /^\d+$/.test(parts[parts.length - 2])) { serialEnd = parts.pop(); serialStart = parts.pop(); factory = parts.pop(); modelType = parts.join(' '); } } } } // 제어기 및 파라미터명 let ecu = mid ? `MID${mid} 제어 유니트` : ''; const mEcu = fullText.match(/MID[\/\s]*컨트롤\s*유니트\s*([^\r\n]+)/i); if (mEcu) ecu = clean(mEcu[1]); let paramName = pid ? `${pidType}${pid}` : ''; const mParam = fullText.match(/(?:PID\/PPID\/SID\/PSID|PID|PPID|SID|PSID)\s*([^\r\n]+)/i); if (mParam) paramName = clean(mParam[1]); let standard = ''; const mStd = fullText.match(/기준\s*([^\r\n]+)/); if (mStd && !mStd[1].includes('고장 모드 진단')) standard = clean(mStd[1]); // FMI 모드 리스트 const fmiMatches = fullText.match(/고장\s*모드\s*FMI\s*(\d+)|FMI\s*(\d+)/gi) || ['FMI 3', 'FMI 5']; const uniqueFmis = Array.from(new Set(fmiMatches.map(m => m.replace(/\D/g, '')).filter(Boolean))); const fmiList = uniqueFmis.map(fmi => { const secRegex = new RegExp("FMI\\s*" + fmi + "([\\s\\S]*?)(?=(?:고장\\s*모드\\s*FMI\\s*\\d+|고장\\s*모드|Guided Diagnostics|$))", "i"); const secMatch = fullText.match(secRegex), secText = secMatch ? secMatch[1] : ''; const mCond = secText.match(/탐지\s*조건:?\s*([\s\S]*?)(?=(?:증상|가능한|조치|$))/i); const mSym = secText.match(/증상:?\s*([\s\S]*?)(?=(?:가능한|조치|$))/i); const mCause = secText.match(/가능한\s*이유:?\s*([\s\S]*?)(?=(?:조치|$))/i); const mAct = secText.match(/조치:?\s*([^\r\n]+)/i); return { fmi: fmi, cond: mCond ? clean(mCond[1]) : '', symptom: mSym ? cleanBullets(mSym[1]) : '없음', causes: mCause ? cleanBullets(mCause[1]) : '없음', actions: mAct ? clean(mAct[1]) : disp }; }); resolve({ docType: 'sae', item: { dtc: dtc, displayMsg: disp, modelType: modelType, factory: factory, serialStart: serialStart, serialEnd: serialEnd, ecu: ecu, paramName: paramName, standard: standard, fmiList: fmiList } }); } else { const dtcMatch = fullText.match(/([P|U|B|C][0-9A-Z]{5,7})/); resolve({ docType: 'iso', item: { dtc: dtcMatch ? dtcMatch[1].replace(/\s+/g, '') : '코드미상', modelType: 'EW140E Volvo', factory: '', serialStart: '', serialEnd: '', ecu: '엔진 제어 모듈', compFull: dtcMatch ? dtcMatch[1] : '', standard: '개요', event: '고장 감지', symptom: '없음', condition: '없음', preDtc: '없음', causes: '없음', actions: '없음' } }); } } catch (err) { reject(err); } }; reader.readAsArrayBuffer(file); }); }
